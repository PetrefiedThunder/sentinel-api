    13	def _is_supported_approver(value: str) -> bool:
    14	    if value.startswith("sms:"):
    15	        return bool(value[len("sms:"):].strip())
    16	    if value.startswith("mailto:"):
    17	        return "@" in value[len("mailto:"):]
    18	    return "@" in value and ":" not in value.split("@", 1)[0]

    59	@router.post("")
    60	async def create(
    61	    payload: ApprovalCreate,
    62	    background_tasks: BackgroundTasks,
    63	    db: AsyncSession = Depends(get_db),
    64	    tenant: Tenant = Depends(get_current_tenant),
    65	    idempotency_key: str | None = Header(default=None, alias="Idempotency-Key"),
    66	):
    67	    """Create an approval. Pass `Idempotency-Key: <opaque>` to make retries
    68	    safe — the same key replays the original response without firing
    69	    duplicate emails or creating duplicate rows."""
    70	
    71	    async def _do_create() -> dict:
    72	        for approver in payload.approvers:
    73	            phone = sms_approver_phone(approver)
    74	            if phone and not await find_active_sms_contact(db, tenant.id, phone):
    75	                raise HTTPException(
    76	                    400, "SMS approver requires active SMS consent contact"
    77	                )
    78	        approval = await create_approval(
    79	            db, tenant, payload, background_tasks=background_tasks
    80	        )
    81	        return {
    82	            "action_id": approval.id,
    83	            "status": approval.decision,
    84	            **_serialize(approval),
    85	        }
    86	
    87	    return await run_with_idempotency(
    88	        db,
    89	        tenant_id=tenant.id,
    90	        idempotency_key=idempotency_key,
    91	        method="POST",
    92	        path="/v1/approvals",
    93	        request_body=payload.model_dump(),
    94	        handler=_do_create,
    95	    )

    98	@router.get("")
    99	async def list_approvals(
   100	    status: str | None = None,
   101	    limit: int | None = Query(None, ge=1, le=100),
   102	    cursor: str | None = Query(None),
   103	    db: AsyncSession = Depends(get_db),
   104	    tenant: Tenant = Depends(get_current_tenant),
   105	):
   106	    """List approvals.
   107	
   108	    Backward-compatible: with no `limit` and no `cursor`, returns the legacy
   109	    top-200 bare array. With either present, returns the cursor-paginated
   110	    envelope {data, has_more, next_cursor}.
   111	    """
   112	    base = select(Approval).where(Approval.tenant_id == tenant.id)
   113	    if status:
   114	        base = base.where(Approval.decision == status)
   115	
   116	    if limit is None and cursor is None:
   117	        # Legacy shape — don't break existing SDK clients
   118	        stmt = base.order_by(Approval.created_at.desc(), Approval.id.desc()).limit(200)
   119	        result = await db.execute(stmt)
   120	        return [_serialize(a) for a in result.scalars().all()]
   121	
   122	    page = await paginate_stmt(db, base, Approval, limit=limit or 50, cursor=cursor)
   123	    return {
   124	        "data": [_serialize(a) for a in page.items],
   125	        "has_more": page.has_more,
   126	        "next_cursor": page.next_cursor,
   127	    }

   136	):

   142	        raise HTTPException(404, "Not found")
   143	    if approval.decision != "pending":
   144	        return _serialize(approval)

   146	    deadline = asyncio.get_event_loop().time() + timeout
   147	    poll_interval = 0.1
   148	    # Race: NOTIFY-wait against a poll loop. Whichever wakes first wins.
   149	    while True:
   150	        remaining = deadline - asyncio.get_event_loop().time()

   133	    }
