    17	async def _verified_form(request: Request):
    18	    form = await request.form()
    19	    params = {key: str(value) for key, value in form.items()}
    20	    signature = request.headers.get("X-Twilio-Signature", "")
    21	    if settings.TWILIO_AUTH_TOKEN:
    22	        validator = RequestValidator(settings.TWILIO_AUTH_TOKEN)
    23	        validation_url = f"{settings.PUBLIC_API_URL.rstrip('/')}{request.url.path}"
    24	        if request.url.query:
    25	            validation_url = f"{validation_url}?{request.url.query}"
    26	        if not validator.validate(validation_url, params, signature):
    27	            raise HTTPException(401, "Invalid Twilio signature")
    28	    return params

    33	def client_ip(request: Request) -> str:
    34	    """Best-effort client IP. Trust the first hop from X-Forwarded-For when set
    35	    by our edge (Vercel proxy / Railway router), else fall back to the socket."""
    36	    fwd = request.headers.get("x-forwarded-for", "")
    37	    if fwd:
    38	        # left-most entry is the original client per RFC 7239
    39	        return fwd.split(",")[0].strip()
    40	    if request.client:
    41	        return request.client.host
    42	    return "unknown"

    45	async def enforce(
    46	    request: Request,
    47	    *,
    48	    bucket: str,
    49	    limit: int,
    50	    window_seconds: int,
    51	    key_suffix: str = "",
    52	    by_ip: bool = True,
    53	) -> None:
    54	    """Throw 429 if more than `limit` requests in `window_seconds`.
    55	
    56	    Args:
    57	        bucket: short identifier of the protected route, e.g. "signup".
    58	        limit: max requests per window.
    59	        window_seconds: window length.
    60	        key_suffix: optional extra discriminator (e.g. lowercased email) so the
    61	            same IP using different emails still gets independent counters.
    62	        by_ip: when True (default), bucket per client IP. Set False to bucket
    63	            globally on (bucket, key_suffix) — useful when edge proxies rotate
    64	            client IPs, or when you want to throttle by email-domain alone.
    65	
    66	    Side effect: stashes the latest (limit, remaining, reset_in) onto
    67	    `request.state.rate_limit_headers` so the response middleware can
    68	    surface X-RateLimit-* headers — Stripe-style ergonomics.
    69	    """
    70	    r = _client()
    71	    if r is None:
    72	        # fail-open: better to allow than 500. Skip header surfacing too.
    73	        return
    74	
    75	    parts = ["rl", bucket]
    76	    if by_ip:
    77	        parts.append(client_ip(request))
    78	    if key_suffix:
    79	        parts.append(key_suffix)
    80	    key = ":".join(parts)
    81	    try:
    82	        count = await r.incr(key)
    83	        if count == 1:
    84	            await r.expire(key, window_seconds)
    85	        ttl_raw = await r.ttl(key)
    86	        ttl = window_seconds if (ttl_raw is None or ttl_raw < 0) else ttl_raw
    87	        remaining = max(0, limit - count)
    88	
    89	        # Stash for response middleware. Last-bucket-checked wins; if a
    90	        # handler enforces several buckets, the tightest is most useful —
    91	        # callers should call the most-restrictive bucket LAST.
    92	        request.state.rate_limit_headers = {
    93	            "X-RateLimit-Limit": str(limit),
    94	            "X-RateLimit-Remaining": str(remaining),
    95	            "X-RateLimit-Reset": str(ttl),
    96	            "X-RateLimit-Bucket": bucket,
    97	        }
    98	
    99	        if count > limit:
   100	            raise HTTPException(
   101	                status_code=429,
   102	                detail=(
   103	                    f"Rate limit exceeded for {bucket}. "
   104	                    f"Try again in {ttl} seconds."
   105	                ),
   106	                headers={
   107	                    "Retry-After": str(ttl),
   108	                    "X-RateLimit-Limit": str(limit),
   109	                    "X-RateLimit-Remaining": "0",
   110	                    "X-RateLimit-Reset": str(ttl),
   111	                    "X-RateLimit-Bucket": bucket,
   112	                },
   113	            )
   114	    except HTTPException:
   115	        raise
   116	    except Exception as e:
   117	        # any other Redis hiccup → fail-open
   118	        log.warning("Rate-limit check failed (fail-open): %s", e)
   119	        return
