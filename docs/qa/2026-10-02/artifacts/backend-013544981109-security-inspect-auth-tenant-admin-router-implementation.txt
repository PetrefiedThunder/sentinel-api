    12	def hash_key(raw: str) -> str:
    13	    return hashlib.sha256(raw.encode()).hexdigest()

    16	def generate_api_key(mode: str = "live") -> tuple[str, str, str]:
    17	    """Issue an API key. mode='live' → sk_live_…, mode='test' → sk_test_…
    18	
    19	    Test-mode keys are scoped to a test workspace whose approvals never fire
    20	    real emails/SMS — see services that gate on tenant.mode.
    21	    """
    22	    token_prefix = "sk_test_" if mode == "test" else "sk_live_"
    23	    raw = f"{token_prefix}{secrets.token_hex(32)}"
    24	    prefix = raw[:11]
    25	    return raw, prefix, hash_key(raw)

    28	async def get_current_tenant(
    29	    authorization: [REDACTED] = Header(...),
    30	    db: AsyncSession = Depends(get_db),
    31	) -> Tenant:
    32	    if not authorization or not authorization.lower().startswith("bearer "):
    33	        raise HTTPException(status_code=401, detail="Missing or invalid authorization header")
    34	    raw_key = authorization.split(" ", 1)[1].strip()
    35	    key_hash = hash_key(raw_key)
    36	    result = await db.execute(
    37	        select(ApiKey).where(ApiKey.key_hash == key_hash, ApiKey.revoked_at.is_(None))
    38	    )
    39	    api_key = result.scalar_one_or_none()
    40	    if not api_key:
    41	        raise HTTPException(status_code=401, detail="Invalid API key")
    42	    tenant = await db.get(Tenant, api_key.tenant_id)
    43	    if not tenant:
    44	        raise HTTPException(status_code=401, detail="Tenant not found")
    45	    # Stamp tenant_id on every subsequent log line in this request
    46	    from app.logging_setup import bind_request_context
    47	    bind_request_context(tenant_id=tenant.id, api_key_id=api_key.id)
    48	    return tenant

    33	@router.post("/signup")
    34	async def signup(
    35	    payload: TenantSignup,
    36	    background: BackgroundTasks,
    37	    request: Request,
    38	    db: AsyncSession = Depends(get_db),
    39	):
    40	    if "@" not in payload.email:
    41	        raise HTTPException(400, "Invalid email")
    42	    domain = payload.email.split("@")[-1].lower()
    43	    if domain in BLOCKED_DOMAINS:
    44	        raise HTTPException(
    45	            400, "Work email required (personal email domains not allowed)"
    46	        )
    47	    # Throttle per IP (5/hour) AND per email-domain (10/hour). Edge proxies
    48	    # rewrite client IPs through internal NAT, so an IP-only limit is leaky;
    49	    # the domain limit catches mass signups against one company regardless of
    50	    # origin IP.
    51	    await rate_limit(request, bucket="signup", limit=5, window_seconds=3600)
    52	    await rate_limit(
    53	        request,
    54	        bucket="signup-domain",
    55	        limit=10,
    56	        window_seconds=3600,
    57	        key_suffix=domain,
    58	        by_ip=False,
    59	    )
    60	    existing = await db.execute(select(Tenant).where(Tenant.email == payload.email))
    61	    if existing.scalar_one_or_none():
    62	        raise HTTPException(400, "Email already registered")
    63	    tenant = Tenant(name=payload.name, email=payload.email, mode=payload.mode)
    64	    db.add(tenant)
    65	    await db.flush()
    66	    raw, prefix, key_hash = generate_api_key(mode=payload.mode)
    67	    api_key = ApiKey(
    68	        tenant_id=tenant.id, key_hash=key_hash, prefix=prefix, name="default"
    69	    )
    70	    db.add(api_key)
    71	    await db.commit()
    72	    await db.refresh(tenant)
    73	
    74	    # Live tenants get the welcome+verify email. Test workspaces don't —
    75	    # they're for integration testing, not a real human signup.
    76	    if tenant.mode == "live":
    77	        background.add_task(send_welcome_email, tenant, raw)
    78	
    79	    return {"tenant_id": tenant.id, "api_key": raw, "mode": tenant.mode}

    87	@router.post("/verify-email")
    88	async def verify_email(payload: VerifyPayload, db: AsyncSession = Depends(get_db)):
    89	    try:
    90	        tenant_id = verify_token(payload.token, expected_purpose="verify")
    91	    except InvalidOnboardingToken as e:
    92	        raise HTTPException(400, f"Invalid or expired verification link: {e}") from e
    93	    tenant = await db.get(Tenant, tenant_id)
    94	    if tenant is None:
    95	        raise HTTPException(404, "Workspace not found")
    96	    if tenant.email_verified_at is None:
    97	        tenant.email_verified_at = datetime.utcnow()
    98	        await db.commit()
    99	    return {
   100	        "tenant_id": tenant.id,
   101	        "email": tenant.email,
   102	        "email_verified_at": tenant.email_verified_at,
   103	    }

   106	# ── API key recovery (magic-link, rotates key) ────────────────────────
   107	class RecoverRequest(BaseModel):
   108	    email: str
   109	
   110	
   111	@router.post("/recover/request")
   112	async def recover_request(
   113	    payload: RecoverRequest,
   114	    background: BackgroundTasks,
   115	    request: Request,
   116	    db: AsyncSession = Depends(get_db),
   117	):
   118	    """Step 1: user enters email → we email them a magic link.
   119	
   120	    We always return 204, regardless of whether the email matches a real
   121	    tenant, so this endpoint cannot be used to enumerate accounts.
   122	    """
   123	    # 3 recovery emails per IP per hour. Throttles harassment + cost-bomb.
   124	    await rate_limit(request, bucket="recover", limit=3, window_seconds=3600)
   125	    email = (payload.email or "").strip().lower()
   126	    if "@" not in email:
   127	        # never reveal whether email exists — always pretend success
   128	        return {"ok": True}
   129	    result = await db.execute(select(Tenant).where(Tenant.email == email))
   130	    tenant = result.scalar_one_or_none()
   131	    if tenant is not None:
   132	        background.add_task(send_recovery_email, tenant)
   133	    return {"ok": True}

   137	    token: str

   139	
   140	@router.post("/recover/exchange")
   141	async def recover_exchange(
   142	    payload: RecoverExchange, db: AsyncSession = Depends(get_db)
   143	):
   144	    """Step 2: dashboard POSTs the token from the magic link. We revoke all
   145	    existing keys and issue a brand-new one. The new key is returned ONCE."""
   146	    try:
   147	        tenant_id = verify_token(payload.token, expected_purpose="recover")
   148	    except InvalidOnboardingToken as e:
   149	        raise HTTPException(400, f"Invalid or expired recovery link: {e}") from e
   150	
   151	    tenant = await db.get(Tenant, tenant_id)
   152	    if tenant is None:
   153	        raise HTTPException(404, "Workspace not found")
   154	
   155	    # revoke every active key on this tenant
   156	    now = datetime.utcnow()
   157	    existing = await db.execute(
   158	        select(ApiKey).where(ApiKey.tenant_id == tenant.id, ApiKey.revoked_at.is_(None))
   159	    )
   160	    for k in existing.scalars():
   161	        k.revoked_at = now
   162	
   163	    # issue a new key
   164	    raw, prefix, key_hash = generate_api_key()
   165	    db.add(
   166	        ApiKey(
   167	            tenant_id=tenant.id, key_hash=key_hash, prefix=prefix, name="recovery"
   168	        )
   169	    )
   170	    await db.commit()
   171	    return {"tenant_id": tenant.id, "api_key": raw}

   196	    tenant: Tenant = Depends(get_current_tenant),
   197	):

   208	    await db.commit()
   209	    await db.refresh(tenant)
   210	    return _serialize_tenant(tenant)

    23	def _check_admin(
    24	    authorization: [REDACTED] | None = Header(default=None),
    25	) -> None:
    26	    """Reject unless Authorization: Bearer [REDACTED] settings.ADMIN_TOKEN."""
    27	    expected = settings.ADMIN_TOKEN
    28	    if not expected:
    29	        raise HTTPException(503, "Admin endpoints disabled (ADMIN_TOKEN not set)")
    30	    if not authorization or not authorization.lower().startswith("bearer "):
    31	        raise HTTPException(401, "Missing admin token")
    32	    token = authorization.split(" ", 1)[1].strip()
    33	    if token != expected:
    34	        raise HTTPException(403, "Invalid admin token")

    57	@router.post("/admin/purge-unverified", dependencies=[Depends(_check_admin)])
    58	async def purge_unverified(
    59	    older_than_days: int = 30,
    60	    dry_run: bool = True,
    61	    db: AsyncSession = Depends(get_db),
    62	) -> dict:
    63	    """Delete tenants that signed up more than `older_than_days` ago and
    64	    never verified their email. Cascades to api_keys + approvals + audit_events.
    65	
    66	    Defaults to dry_run=True — pass dry_run=false to actually delete.
    67	    """
    68	    if older_than_days < 1:
    69	        raise HTTPException(400, "older_than_days must be >= 1")
    70	
    71	    cutoff = datetime.utcnow() - timedelta(days=older_than_days)
    72	    q = await db.execute(
    73	        select(Tenant).where(
    74	            and_(
    75	                Tenant.email_verified_at.is_(None),
    76	                Tenant.created_at < cutoff,
    77	            )
    78	        )
    79	    )
    80	    targets = list(q.scalars())
    81	
    82	    sample = [{"id": t.id, "email": t.email, "created_at": str(t.created_at)} for t in targets[:5]]
    83	
    84	    if dry_run:
    85	        return {
    86	            "dry_run": True,
    87	            "older_than_days": older_than_days,
    88	            "cutoff": cutoff.isoformat(),
    89	            "would_delete": len(targets),
    90	            "sample": sample,
    91	        }
    92	
    93	    deleted_keys = 0
    94	    deleted_approvals = 0
    95	    deleted_audit = 0
    96	    for t in targets:
    97	        # delete dependent rows first to avoid FK violations
    98	        r = await db.execute(delete(AuditEvent).where(AuditEvent.tenant_id == t.id))
    99	        deleted_audit += r.rowcount or 0
   100	        r = await db.execute(delete(ApiKey).where(ApiKey.tenant_id == t.id))
   101	        deleted_keys += r.rowcount or 0
   102	        r = await db.execute(delete(Approval).where(Approval.tenant_id == t.id))
   103	        deleted_approvals += r.rowcount or 0
   104	        await db.execute(delete(Tenant).where(Tenant.id == t.id))
   105	    await db.commit()
   106	
   107	    return {
   108	        "dry_run": False,
   109	        "older_than_days": older_than_days,
   110	        "deleted_tenants": len(targets),
   111	        "deleted_api_keys": deleted_keys,
   112	        "deleted_approvals": deleted_approvals,
   113	        "deleted_audit_events": deleted_audit,
   114	        "sample": sample,
   115	    }
