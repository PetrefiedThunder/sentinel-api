    43	    db: AsyncSession = Depends(get_db),
    44	    tenant: Tenant = Depends(get_current_tenant),
    45	):

    92	@router.get("/verify")
    93	async def verify_chain(
    94	    db: AsyncSession = Depends(get_db),
    95	    tenant: Tenant = Depends(get_current_tenant),
    96	):
    97	    """Walk the tenant's full audit hash chain and prove tamper-evidence.
    98	
    99	    Recomputes each event's hash from its stored payload (same computation as
   100	    app/services/audit_log.py) and checks the prev_hash linkage. Read-only.
   101	    """
   102	    stmt = (
   103	        select(AuditEvent)
   104	        .where(AuditEvent.tenant_id == tenant.id)
   105	        .order_by(AuditEvent.created_at.asc(), AuditEvent.id.asc())
   106	    )
   107	    result = await db.execute(stmt)
   108	    events = result.scalars().all()
   109	
   110	    valid = True
   111	    first_invalid_event_id = None
   112	    prev_event_hash = None
   113	
   114	    for e in events:
   115	        # Same payload shape append_audit_event hashes at write time.
   116	        payload = {
   117	            "action_id": e.action_id,
   118	            "execution_result": e.execution_result,
   119	            "error": e.error,
   120	        }
   121	        expected_hash = _compute_hash(e.prev_hash, payload)
   122	        if e.event_hash != expected_hash or e.prev_hash != prev_event_hash:
   123	            valid = False
   124	            first_invalid_event_id = e.id
   125	            break
   126	        prev_event_hash = e.event_hash
   127	
   128	    return {
   129	        "valid": valid,
   130	        "events_checked": len(events),
   131	        "first_invalid_event_id": first_invalid_event_id,
   132	        "checked_at": datetime.now(UTC).isoformat(),
   133	    }

     9	def _compute_hash(prev_hash, payload):
    10	    serialized = json.dumps(payload, sort_keys=True, default=str)
    11	    data = (prev_hash or "") + serialized
    12	    return hashlib.sha256(data.encode()).hexdigest()

    15	async def append_audit_event(
    16	    db, tenant_id, action_id, execution_result, error=None, *, commit=True
    17	):
    18	    # Serialize concurrent appends per tenant. Without this, two simultaneous
    19	    # appends both read the same "previous" event and fork the chain — found
    20	    # in production by GET /v1/audit-events/verify on 2026-06-10 (two forks
    21	    # left by the 5-way-concurrent approval test). The Postgres advisory lock
    22	    # is held until the surrounding transaction commits; sqlite (tests) is
    23	    # single-writer so the lock is unnecessary there.
    24	    bind = db.get_bind()
    25	    if bind is not None and bind.dialect.name == "postgresql":
    26	        await db.execute(
    27	            text("SELECT pg_advisory_xact_lock(hashtext(:tid))"),
    28	            {"tid": tenant_id},
    29	        )
    30	
    31	    result = await db.execute(
    32	        select(AuditEvent)
    33	        .where(AuditEvent.tenant_id == tenant_id)
    34	        # id tiebreak matters: verify walks (created_at asc, id asc), so the
    35	        # chain head must be the max of that same ordering.
    36	        .order_by(desc(AuditEvent.created_at), desc(AuditEvent.id))
    37	        .limit(1)
    38	    )
    39	    prev = result.scalar_one_or_none()
    40	    prev_hash = prev.event_hash if prev else None
    41	
    42	    payload = {
    43	        "action_id": action_id,
    44	        "execution_result": execution_result,
    45	        "error": error,
    46	    }
    47	    event_hash = _compute_hash(prev_hash, payload)
    48	
    49	    event = AuditEvent(
    50	        tenant_id=tenant_id,
    51	        action_id=action_id,
    52	        execution_result=execution_result,
    53	        error=error,
    54	        prev_hash=prev_hash,
    55	        event_hash=event_hash,
    56	    )
    57	    db.add(event)
    58	    if commit:
    59	        await db.commit()
    60	        await db.refresh(event)
    61	    else:
    62	        # Decision routes own the transaction containing their audit evidence.
    63	        await db.flush()
    64	    return event

     1	"""Webhook delivery — HMAC-signed POSTs to customer URLs on approval decisions.
     2	
     3	Flow:
     4	  1. Customer registers an endpoint via POST /v1/webhooks with a URL.
     5	     We mint a random secret and return it ONCE.
     6	  2. When an approval is decided (approved/rejected), we look up matching
     7	     endpoints for that tenant and fire a POST.
     8	  3. The POST body is JSON: {event, action_id, decision, decided_by, ...}
     9	  4. We sign the body with HMAC-SHA256(secret, raw_body) and put the
    10	     signature in the `X-Sentinel-Signature` header. Customer can verify.
    11	  5. We log every attempt in webhook_deliveries (status_code, response_snippet).
    12	  6. Best-effort with 3 retries on 5xx/timeout, exponential backoff
    13	     (1s, 4s, 16s). Never blocks the decision endpoint.
    14	"""
    15	from __future__ import annotations
    16	
    17	import asyncio
    18	import hashlib
    19	import hmac
    20	import json
    21	import logging
    22	import secrets
    23	from datetime import datetime
    24	from typing import Any
    25	
    26	import httpx
    27	from sqlalchemy import select
    28	from sqlalchemy.ext.asyncio import AsyncSession
    29	
    30	from app.models import Approval, WebhookDelivery, WebhookEndpoint
    31	
    32	log = logging.getLogger(__name__)
    33	
    34	MAX_ATTEMPTS = 3
    35	BACKOFF_SECONDS = [1.0, 4.0, 16.0]
    36	HTTP_TIMEOUT = 8.0
    37	SIGNATURE_HEADER = "X-Sentinel-Signature"
    38	EVENT_HEADER = "X-Sentinel-Event"
    39	ID_HEADER = "X-Sentinel-Delivery"
    40	
    41	# The event loop only keeps weak references to tasks; without a strong
    42	# reference here, an in-flight delivery (and its retries) can be GC'd.
    43	_inflight: set[asyncio.Task] = set()
    44	
    45	
    46	def _track(task: asyncio.Task) -> None:
    47	    _inflight.add(task)
    48	    task.add_done_callback(_on_delivery_done)
    49	
    50	
    51	def _on_delivery_done(task: asyncio.Task) -> None:
    52	    _inflight.discard(task)
    53	    if not task.cancelled() and (exc := task.exception()) is not None:
    54	        log.error("webhook delivery task crashed: %s", exc, exc_info=exc)

    57	def generate_secret() -> str:
    58	    """64-char URL-safe random secret. Shown ONCE on creation."""
    59	    return f"whsec_{secrets.token_urlsafe(32)}"
    60	
    61	
    62	def sign_body(secret: str, raw_body: bytes) -> str:
    63	    """HMAC-SHA256 hex digest of the raw POST body, signed with the endpoint secret.
    64	
    65	    Customer verifies via:
    66	        expected = hmac.new(secret.encode(), raw_body, sha256).hexdigest()
    67	        if not hmac.compare_digest(expected, request.headers["X-Sentinel-Signature"]):
    68	            return 401
    69	    """
    70	    return hmac.new(secret.encode(), raw_body, hashlib.sha256).hexdigest()

   105	
   106	async def dispatch_approval_webhook(
   107	    db: AsyncSession, approval: Approval
   108	) -> None:
   109	    """Look up all webhook endpoints for this tenant subscribed to the event,
   110	    fire each one as a background task (best-effort with retries)."""
   111	    event_type = _event_type_for(approval)
   112	    if event_type is None:
   113	        return
   114	
   115	    result = await db.execute(
   116	        select(WebhookEndpoint).where(
   117	            WebhookEndpoint.tenant_id == approval.tenant_id,
   118	            WebhookEndpoint.disabled_at.is_(None),
   119	        )
   120	    )
   121	    endpoints = [e for e in result.scalars() if _matches_filter(e, event_type)]
   122	    if not endpoints:
   123	        return
   124	
   125	    for endpoint in endpoints:
   126	        # Spawn each delivery as its own task — never block the caller.
   127	        _track(asyncio.create_task(_deliver_with_retries(endpoint, event_type, approval)))
   128	
   129	
   130	async def _deliver_with_retries(
   131	    endpoint: WebhookEndpoint, event_type: str, approval: Approval
   132	) -> None:
   133	    """Run in a background task. Owns its own DB session to avoid lifecycle issues."""
   134	    from app.db import SessionLocal
   135	
   136	    delivery_id = f"whd_{secrets.token_hex(8)}"
   137	    payload = _build_payload(event_type, approval, delivery_id)
   138	    raw_body = json.dumps(payload, separators=(",", ":"), sort_keys=True).encode()
   139	    signature = sign_body(endpoint.secret, raw_body)
   140	    headers = {
   141	        "Content-Type": "application/json",
   142	        "User-Agent": "Sentinel-Webhook/1.0",
   143	        SIGNATURE_HEADER: signature,
   144	        EVENT_HEADER: event_type,
   145	        ID_HEADER: delivery_id,
   146	    }
   147	
   148	    last_status: int | None = None
   149	    last_snippet: str | None = None
   150	    success = False
   151	
   152	    async with httpx.AsyncClient(timeout=HTTP_TIMEOUT) as client:
   153	        for attempt in range(1, MAX_ATTEMPTS + 1):
   154	            try:
   155	                r = await client.post(endpoint.url, content=raw_body, headers=headers)
   156	                last_status = r.status_code
   157	                last_snippet = (r.text or "")[:512]
   158	                if 200 <= r.status_code < 300:
   159	                    success = True
   160	                    break
   161	                # 4xx (except 408/429) — don't retry, the customer rejected us
   162	                if 400 <= r.status_code < 500 and r.status_code not in (408, 429):
   163	                    break
   164	            except httpx.RequestError as e:
   165	                last_snippet = f"{type(e).__name__}: {str(e)[:400]}"
   166	            if attempt < MAX_ATTEMPTS:
   167	                await asyncio.sleep(BACKOFF_SECONDS[attempt - 1])
   168	
   169	    # Record delivery + bump endpoint last-used/status
   170	    try:
   171	        async with SessionLocal() as db2:
   172	            db2.add(
   173	                WebhookDelivery(
   174	                    id=delivery_id,
   175	                    endpoint_id=endpoint.id,
   176	                    tenant_id=endpoint.tenant_id,
   177	                    event_type=event_type,
   178	                    action_id=approval.id,
   179	                    payload=payload,
   180	                    attempt=attempt,
   181	                    status_code=last_status,
   182	                    response_snippet=last_snippet,
   183	                    delivered_at=datetime.utcnow() if success else None,
   184	                )
   185	            )
   186	            ep = await db2.get(WebhookEndpoint, endpoint.id)
   187	            if ep is not None:
   188	                ep.last_used_at = datetime.utcnow()
   189	                ep.last_status_code = last_status
   190	            await db2.commit()
   191	    except Exception:
   192	        log.exception("Failed to record webhook delivery for endpoint %s", endpoint.id)

     3	Uses the same HMAC-signed token primitive as approval magic links
     4	(see approval_tokens.py) so we share the same JWT_SECRET infra.

    22	# ── HMAC token (purpose-scoped) ──────────────────────────────────────
    23	def _sign(payload_part: str) -> str:
    24	    digest = hmac.new(
    25	        settings.JWT_SECRET.encode(),
    26	        payload_part.encode(),
    27	        hashlib.sha256,
    28	    ).digest()
    29	    return base64.urlsafe_b64encode(digest).decode().rstrip("=")
    30	
    31	
    32	def _b64e(b: bytes) -> str:
    33	    return base64.urlsafe_b64encode(b).decode().rstrip("=")
    34	
    35	
    36	def _b64d(s: str) -> bytes:
    37	    return base64.urlsafe_b64decode(s + "=" * (-len(s) % 4))
    38	
    39	
    40	class InvalidOnboardingToken(ValueError):
    41	    pass
    42	
    43	
    44	def create_token(purpose: str, subject: str, ttl_seconds: int) -> str:
    45	    """Signed token: base64(json({purpose, sub, exp})) + '.' + sig."""
    46	    payload = {
    47	        "p": purpose,           # "verify" | "recover"
    48	        "sub": subject,         # tenant.id
    49	        "exp": int(time.time()) + int(ttl_seconds),
    50	    }
    51	    payload_part = _b64e(
    52	        json.dumps(payload, sort_keys=True, separators=(",", ":")).encode()
    53	    )
    54	    return f"{payload_part}.{_sign(payload_part)}"
    55	
    56	
    57	def verify_token(token: str, expected_purpose: str) -> str:
    58	    """Return subject (tenant_id) if the token is valid for `expected_purpose`."""
    59	    try:
    60	        payload_part, signature_part = token.split(".", 1)
    61	    except ValueError as e:
    62	        raise InvalidOnboardingToken("Malformed token") from e
    63	    if not hmac.compare_digest(_sign(payload_part), signature_part):
    64	        raise InvalidOnboardingToken("Invalid signature")
    65	    try:
    66	        payload = json.loads(_b64d(payload_part))
    67	    except (ValueError, TypeError) as e:
    68	        raise InvalidOnboardingToken("Malformed payload") from e
    69	    if payload.get("p") != expected_purpose:
    70	        raise InvalidOnboardingToken("Wrong purpose")
    71	    if int(payload.get("exp", 0)) < int(time.time()):
    72	        raise InvalidOnboardingToken("Expired")
    73	    sub = payload.get("sub")
    74	    if not sub or not isinstance(sub, str):
    75	        raise InvalidOnboardingToken("Missing subject")
    76	    return sub
