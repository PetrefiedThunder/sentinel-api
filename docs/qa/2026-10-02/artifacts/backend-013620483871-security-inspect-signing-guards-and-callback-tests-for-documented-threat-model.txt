    23	def create_decision_token(
    24	    action_id: str,
    25	    decision: str,
    26	    expires_in_seconds: int,
    27	    now: int | None = None,
    28	) -> str:
    29	    if decision not in VALID_DECISIONS:
    30	        raise ValueError("decision must be 'approved' or 'rejected'")
    31	    issued_at = int(time.time() if now is None else now)
    32	    payload = {
    33	        "action_id": action_id,
    34	        "decision": decision,
    35	        "exp": issued_at + expires_in_seconds,
    36	    }
    37	    payload_part = _base64url_encode(
    38	        json.dumps(payload, sort_keys=True, separators=(",", ":")).encode()
    39	    )
    40	    signature_part = _sign(payload_part)
    41	    return f"{payload_part}.{signature_part}"

    44	def verify_decision_token(
    45	    token: str,
    46	    action_id: str,
    47	    now: int | None = None,
    48	) -> str:
    49	    try:
    50	        payload_part, signature_part = token.split(".", 1)
    51	    except ValueError as exc:
    52	        raise InvalidApprovalToken("Malformed approval token") from exc
    53	
    54	    expected_signature = _sign(payload_part)
    55	    if not hmac.compare_digest(expected_signature, signature_part):
    56	        raise InvalidApprovalToken("Invalid approval token signature")
    57	
    58	    try:
    59	        payload = json.loads(_base64url_decode(payload_part))
    60	    except (ValueError, TypeError) as exc:
    61	        raise InvalidApprovalToken("Malformed approval token payload") from exc
    62	
    63	    if payload.get("action_id") != action_id:
    64	        raise InvalidApprovalToken("Approval token action mismatch")
    65	    decision = payload.get("decision")
    66	    if decision not in VALID_DECISIONS:
    67	        raise InvalidApprovalToken("Invalid approval token decision")
    68	    current_time = int(time.time() if now is None else now)
    69	    if int(payload.get("exp", 0)) < current_time:
    70	        raise InvalidApprovalToken("Approval token expired")
    71	    return decision

    74	def _sign(payload_part: str) -> str:
    75	    _require_strong_signing_secret()
    76	    digest = hmac.new(
    77	        settings.JWT_SECRET.encode(),
    78	        payload_part.encode(),
    79	        hashlib.sha256,
    80	    ).digest()
    81	    return _base64url_encode(digest)
    82	
    83	
    84	def _base64url_encode(value: bytes) -> str:
    85	    return base64.urlsafe_b64encode(value).decode().rstrip("=")
    86	
    87	
    88	def _base64url_decode(value: str) -> bytes:
    89	    padding = "=" * (-len(value) % 4)
    90	    return base64.urlsafe_b64decode(value + padding)
    91

    93	def _require_strong_signing_secret() -> None:
    94	    secret = settings.JWT_SECRET
    95	    if secret in WEAK_SIGNING_SECRETS or len(secret) < 32:
    96	        raise RuntimeError("JWT_SECRET must be a strong random value for approval links")

     1	"""Structured (JSON) logging via structlog.
     2	
     3	Configures structlog to render every log line as a single JSON object
     4	with timestamp, level, logger, event, plus any bound context (request_id,
     5	tenant_id, route, etc.). Wired into FastAPI's request lifecycle via
     6	RequestContextMiddleware.
     7	
     8	Why JSON: Railway's log search is grep over raw text. JSON lines let us
     9	filter by `tenant_id` / `route` / `request_id` without false positives.
    10	When we eventually ship logs to Datadog/Better Stack/etc, the JSON
    11	fields become indexed dimensions for free.
    12	"""
    13	from __future__ import annotations
    14	
    15	import logging
    16	import secrets
    17	import sys
    18	from contextvars import ContextVar
    19	from typing import Any
    20	
    21	import structlog
    22	from starlette.middleware.base import BaseHTTPMiddleware
    23	from starlette.requests import Request
    24	from starlette.responses import Response
    25	
    26	# Per-request context that gets stamped onto every log line emitted during
    27	# the request. set() in middleware, .get() inside structlog processor.
    28	_request_ctx: ContextVar[dict[str, Any] | None] = ContextVar("_request_ctx", default=None)
    29	
    30	
    31	def _inject_request_ctx(_, __, event_dict: dict) -> dict:
    32	    """structlog processor: merge any active request context into the log."""
    33	    ctx = _request_ctx.get() or {}
    34	    for k, v in ctx.items():
    35	        event_dict.setdefault(k, v)
    36	    return event_dict
    37

    39	def configure_logging(level: str = "INFO") -> None:
    40	    """Replace stdlib logging output with JSON structlog. Idempotent."""
    41	    timestamper = structlog.processors.TimeStamper(fmt="iso", utc=True)
    42	
    43	    shared_processors = [
    44	        structlog.contextvars.merge_contextvars,
    45	        structlog.stdlib.add_log_level,
    46	        _inject_request_ctx,
    47	        timestamper,
    48	        structlog.processors.StackInfoRenderer(),
    49	        structlog.processors.format_exc_info,
    50	    ]
    51	
    52	    structlog.configure(
    53	        processors=shared_processors + [structlog.processors.JSONRenderer()],
    54	        wrapper_class=structlog.make_filtering_bound_logger(
    55	            logging.getLevelName(level.upper()) if isinstance(level, str) else level
    56	        ),
    57	        logger_factory=structlog.PrintLoggerFactory(file=sys.stdout),
    58	        cache_logger_on_first_use=True,
    59	    )
    60	
    61	    # Route stdlib logging (FastAPI / uvicorn / SQLAlchemy) through structlog
    62	    # so we get JSON for those too — otherwise we'd have two log formats.
    63	    formatter = structlog.stdlib.ProcessorFormatter(
    64	        foreign_pre_chain=shared_processors,
    65	        processors=[
    66	            structlog.stdlib.ProcessorFormatter.remove_processors_meta,
    67	            structlog.processors.JSONRenderer(),
    68	        ],
    69	    )
    70	    handler = logging.StreamHandler(sys.stdout)
    71	    handler.setFormatter(formatter)
    72	
    73	    root = logging.getLogger()
    74	    # Don't double up handlers if reload re-imports this module
    75	    root.handlers = [handler]
    76	    root.setLevel(level.upper())
    77	
    78	    # Quieter health-check noise from uvicorn access logs
    79	    logging.getLogger("uvicorn.access").setLevel(logging.WARNING)

    82	class RequestContextMiddleware(BaseHTTPMiddleware):
    83	    """Stamp every log line emitted during a request with request_id +
    84	    method + route, and emit a structured access log on completion.
    85	
    86	    Tenant_id is added downstream by the auth dependency when a request
    87	    hits an authenticated route — see auth.get_current_tenant.
    88	    """
    89	
    90	    async def dispatch(self, request: Request, call_next) -> Response:
    91	        request_id = request.headers.get("x-request-id") or secrets.token_hex(8)
    92	        ctx = {
    93	            "request_id": request_id,
    94	            "method": request.method,
    95	            "route": request.url.path,
    96	        }
    97	        token = _request_ctx.set(ctx)
    98	        logger = structlog.get_logger("http")
    99	        try:
   100	            response = await call_next(request)
   101	        except Exception:
   102	            logger.exception("request.error")
   103	            raise
   104	        else:
   105	            logger.info(
   106	                "request.completed",
   107	                status_code=response.status_code,
   108	            )
   109	            response.headers["X-Request-Id"] = request_id
   110	            # Surface rate-limit context if the handler enforced a bucket
   111	            rl = getattr(request.state, "rate_limit_headers", None)
   112	            if isinstance(rl, dict):
   113	                for k, v in rl.items():
   114	                    response.headers[k] = v
   115	            return response
   116	        finally:
   117	            _request_ctx.reset(token)
   118	
   119	
   120	def bind_request_context(**fields: Any) -> None:
   121	    """Call from anywhere inside a request handler to attach additional
   122	    fields (tenant_id, action_id, etc.) to every subsequent log line in
   123	    this request."""
   124	    current = dict(_request_ctx.get() or {})
   125	    current.update({k: v for k, v in fields.items() if v is not None})
   126	    _request_ctx.set(current)
   127	
   128	
   129	def get_logger(name: str = "sentinel") -> structlog.stdlib.BoundLogger:
   130	    return structlog.get_logger(name)

    13	def _is_supported_approver(value: str) -> bool:
    14	    if value.startswith("sms:"):
    15	        return bool(value[len("sms:"):].strip())
    16	    if value.startswith("mailto:"):
    17	        return "@" in value[len("mailto:"):]
    18	    return "@" in value and ":" not in value.split("@", 1)[0]
