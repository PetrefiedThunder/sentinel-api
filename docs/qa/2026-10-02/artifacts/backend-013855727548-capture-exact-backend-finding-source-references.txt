    44	def create_token(purpose: str, subject: str, ttl_seconds: int) -> str:
    45	    """Signed token: base64(json({purpose, sub, exp})) + '.' + sig."""
    46	    payload = {
    47	        "p": purpose,           # "verify" | "recover"
    48	        "sub": subject,         # tenant.id
    49	        "exp": int(time.time()) + int(ttl_seconds),
    50	    }
    51	    payload_part = _b64e(
    52	        json.dumps(payload, sort_keys=True, separators=(",", ":")).encode()
    53	    )
    54	    return f"{payload_part}.{_sign(payload_part)}"
