    95	# ── Stripe webhook (subscription events) ─────────────────────────────
    96	@router.post("/webhook", include_in_schema=False)
    97	async def stripe_webhook(
    98	    request: Request,
    99	    stripe_signature: str | None = Header(default=None, alias="Stripe-Signature"),
   100	    db: AsyncSession = Depends(get_db),
   101	):
   102	    """Receive subscription lifecycle events from Stripe. Signature-verified
   103	    via the STRIPE_WEBHOOK_SECRET. Updates tenant.plan on success."""
   104	    _stripe_enabled()
   105	    if not settings.STRIPE_WEBHOOK_SECRET:
   106	        raise HTTPException(503, "STRIPE_WEBHOOK_SECRET not configured")
   107	    if not stripe_signature:
   108	        raise HTTPException(401, "Missing Stripe-Signature header")
   109	
   110	    raw_body = await request.body()
   111	    if not _verify_stripe_signature(
   112	        raw_body, stripe_signature, settings.STRIPE_WEBHOOK_SECRET
   113	    ):
   114	        raise HTTPException(401, "Invalid Stripe signature")
   115	
   116	    try:
   117	        event = httpx.Response(200, content=raw_body).json()
   118	    except Exception:
   119	        raise HTTPException(400, "Malformed JSON body") from None
   120	
   121	    event_type = event.get("type", "")
   122	    obj = (event.get("data") or {}).get("object") or {}
   123	    tenant_id = (
   124	        obj.get("client_reference_id")
   125	        or (obj.get("metadata") or {}).get("tenant_id")
   126	        or None
   127	    )
   128	
   129	    if not tenant_id:
   130	        # try to find by stripe_customer_id (subscription.updated events
   131	        # don't always carry client_reference_id)
   132	        customer_id = obj.get("customer")
   133	        if customer_id:
   134	            r = await db.execute(
   135	                select(Tenant).where(Tenant.stripe_customer_id == customer_id)
   136	            )
   137	            tenant = r.scalar_one_or_none()
   138	            if tenant:
   139	                tenant_id = tenant.id
   140	
   141	    if not tenant_id:
   142	        log.warning("Stripe webhook %s — no tenant_id resolvable", event_type)
   143	        return {"ok": True, "ignored": True}
   144	
   145	    tenant = await db.get(Tenant, tenant_id)
   146	    if not tenant:
   147	        log.warning("Stripe webhook %s — tenant %s not found", event_type, tenant_id)
   148	        return {"ok": True, "ignored": True}
   149	
   150	    now = datetime.utcnow()
   151	
   152	    if event_type in ("checkout.session.completed", "customer.subscription.created"):
   153	        tenant.plan = "pro"
   154	        if obj.get("customer"):
   155	            tenant.stripe_customer_id = obj["customer"]
   156	        if obj.get("subscription"):
   157	            tenant.stripe_subscription_id = obj["subscription"]
   158	        elif obj.get("id") and event_type == "customer.subscription.created":
   159	            tenant.stripe_subscription_id = obj["id"]
   160	        tenant.plan_updated_at = now
   161	
   162	    elif event_type == "customer.subscription.updated":
   163	        status = obj.get("status")
   164	        if status in ("active", "trialing"):
   165	            tenant.plan = "pro"
   166	        elif status in ("past_due", "unpaid", "incomplete"):
   167	            # Don't downgrade automatically — let dunning handle it.
   168	            pass
   169	        elif status in ("canceled", "incomplete_expired"):
   170	            tenant.plan = "free"
   171	        tenant.plan_updated_at = now
   172	
   173	    elif event_type == "customer.subscription.deleted":
   174	        tenant.plan = "free"
   175	        tenant.stripe_subscription_id = None
   176	        tenant.plan_updated_at = now
   177	
   178	    await db.commit()
   179	    return {"ok": True, "event": event_type, "tenant_id": tenant.id, "plan": tenant.plan}

   182	def _verify_stripe_signature(
   183	    raw_body: bytes, sig_header: str, secret: str, tolerance: int = 300
   184	) -> bool:
   185	    """Verify Stripe's v1 webhook signature. Format: 't=TIMESTAMP,v1=SIG[,...]'.
   186	
   187	    Manual implementation so we don't take a hard dep on the stripe SDK.
   188	    Standard pattern; matches stripe.Webhook.construct_event().
   189	    """
   190	    import hashlib
   191	    import hmac
   192	    import time as _time
   193	
   194	    try:
   195	        parts = dict(p.split("=", 1) for p in sig_header.split(","))
   196	    except ValueError:
   197	        return False
   198	    ts = parts.get("t")
   199	    v1 = parts.get("v1")
   200	    if not ts or not v1:
   201	        return False
   202	
   203	    # tolerance to mitigate replay
   204	    try:
   205	        if abs(_time.time() - int(ts)) > tolerance:
   206	            return False
   207	    except ValueError:
   208	        return False
   209	
   210	    signed_payload = f"{ts}.".encode() + raw_body
   211	    expected = hmac.new(secret.encode(), signed_payload, hashlib.sha256).hexdigest()
   212	    return hmac.compare_digest(expected, v1)

    46	@router.get("/history")
    47	async def history(
    48	    days: int = Query(default=30, ge=1, le=92),
    49	    db: AsyncSession = Depends(get_db),
    50	) -> list[dict]:
    51	    """Daily uptime aggregates for the last `days` days, ascending by date.
    52	
    53	    Aggregation happens in Python — probe volume is tiny (a few per minute
    54	    at most) and this sidesteps SQLite/Postgres date-trunc dialect drift.
    55	    """
    56	    cutoff = datetime.utcnow() - timedelta(days=days)
    57	    rows = (
    58	        await db.execute(select(UptimeProbe).where(UptimeProbe.created_at >= cutoff))
    59	    ).scalars()
    60	
    61	    buckets: dict[str, dict] = {}
    62	    for probe in rows:
    63	        day = probe.created_at.strftime("%Y-%m-%d")
    64	        bucket = buckets.setdefault(day, {"probes": 0, "ok": 0})
    65	        bucket["probes"] += 1
    66	        if probe.ok:
    67	            bucket["ok"] += 1
    68	
    69	    return [
    70	        {
    71	            "date": day,
    72	            "probes": bucket["probes"],
    73	            "ok": bucket["ok"],
    74	            "uptime_pct": round(bucket["ok"] / bucket["probes"] * 100, 3),
    75	        }
    76	        for day, bucket in sorted(buckets.items())
    77	    ]
